# ---------------------------------------------------------------- utilidades
def _norm(t):
    """Minúsculas y sin acentos: 'senales' encuentra 'SEÑALES'."""
    t = unicodedata.normalize("NFD", str(t or "").lower())
    return "".join(c for c in t if unicodedata.category(c) != "Mn")


def _txt(v):
    if v is None:
        return ""
    if isinstance(v, (datetime, date)):
        return v.strftime("%d/%m/%Y")
    if isinstance(v, float) and v.is_integer():
        v = int(v)
    return str(v).strip()


def _ahora():
    return datetime.now().strftime("%d/%m/%Y %H:%M")


def _hoy():
    return datetime.now().strftime("%d/%m/%Y")


def parse_fecha(v):
    """'13/05/2026', '13/5/26', datetime o date -> date; vacío o ilegible -> None."""
    if isinstance(v, datetime):
        return v.date()
    if isinstance(v, date):
        return v
    m = re.fullmatch(r"\s*(\d{1,2})[/.-](\d{1,2})[/.-](\d{2}|\d{4})\s*", str(v or ""))
    if not m:
        return None
    d, mes, a = int(m.group(1)), int(m.group(2)), int(m.group(3))
    a += 2000 if a < 100 else 0
    try:
        return date(a, mes, d)
    except ValueError:
        return None


def norm_fecha(v):
    """Texto de fecha normalizado a dd/mm/aaaa ('' si está vacío). ValueError si no se entiende."""
    if str(v or "").strip() == "":
        return ""
    f = parse_fecha(v)
    if not f:
        raise ValueError(f"Fecha no válida: «{v}». Usá el formato dd/mm/aaaa.")
    return f.strftime("%d/%m/%Y")


def _parche_openpyxl():
    """Excel guarda los gráficos de anillo con «tamaño del agujero» 0, que openpyxl rechaza: sin este parche
    no se puede ni abrir un Excel que tenga esos gráficos."""
    try:
        from openpyxl.chart.pie_chart import DoughnutChart
        DoughnutChart.holeSize.min = 0
    except Exception:  # noqa: BLE001
        pass


def cargar_libro(ruta, **kw):
    from openpyxl import load_workbook
    _parche_openpyxl()
    return load_workbook(ruta, **kw)


# ---------------------------------------------------------------- ciclos de corrección (DPC)
def ciclo(con, doc_id, rev):
    return con.execute("SELECT * FROM ciclos WHERE documento_id=? AND rev=?", (doc_id, _txt(rev))).fetchone()


def ciclos_doc(con, doc_id):
    return con.execute("SELECT * FROM ciclos WHERE documento_id=? ORDER BY rev", (doc_id,)).fetchall()


def guardar_ciclo(con, doc_id, rev, devolucion, reenvio, pisar=True):
    """Fechas de un ciclo de corrección: el documento volvió con DPC en la revisión `rev` (fecha de devolución)
    y se reenvió corregido (fecha de reenvío). Con pisar=False solo completa lo que está vacío."""
    rev = _txt(rev)
    devolucion, reenvio = norm_fecha(devolucion), norm_fecha(reenvio)
    actual = ciclo(con, doc_id, rev)
    if actual is None:
        if devolucion or reenvio:
            con.execute("INSERT INTO ciclos(documento_id,rev,fecha_devolucion,fecha_reenvio) VALUES (?,?,?,?)",
                        (doc_id, rev, devolucion, reenvio))
    else:
        dev = devolucion if (pisar or not actual["fecha_devolucion"]) and (devolucion or pisar) else actual["fecha_devolucion"]
        rei = reenvio if (pisar or not actual["fecha_reenvio"]) and (reenvio or pisar) else actual["fecha_reenvio"]
        con.execute("UPDATE ciclos SET fecha_devolucion=?, fecha_reenvio=? WHERE documento_id=? AND rev=?",
                    (dev, rei, doc_id, rev))
    con.commit()


def conectar(path=None):
    path = Path(path or DB_PATH)
    if path.exists():  # base de una versión anterior (sin números de revisión de lista)
        old = sqlite3.connect(path)
        cols = [r[1] for r in old.execute("PRAGMA table_info(historial)")]
        old.close()
        if cols and "lista_n" not in cols:
            try:
                path.replace(path.with_suffix(".db.bak"))
            except PermissionError:
                raise SystemExit("Cerrá la versión anterior de la app (está usando documentos.db) y volvé a abrir.")
    con = sqlite3.connect(path)
    con.row_factory = sqlite3.Row
    con.create_function("norm", 1, _norm)
    con.execute("PRAGMA foreign_keys = ON")
    con.executescript(SCHEMA)
    if "nuestro" not in [r[1] for r in con.execute("PRAGMA table_info(documentos)")]:
        con.execute("ALTER TABLE documentos ADD COLUMN nuestro INTEGER NOT NULL DEFAULT 1")
        con.commit()
    if "archivo_excel" not in [r[1] for r in con.execute("PRAGMA table_info(proyectos)")]:
        con.execute("ALTER TABLE proyectos ADD COLUMN archivo_excel TEXT NOT NULL DEFAULT ''")
        con.commit()
    for col in ("fecha_real", "transmittal"):  # KPI de cumplimiento de entrega
        if col not in [r[1] for r in con.execute("PRAGMA table_info(documentos)")]:
            con.execute(f"ALTER TABLE documentos ADD COLUMN {col} TEXT NOT NULL DEFAULT ''")
            con.commit()
    con.execute("INSERT OR IGNORE INTO clientes(nombre) SELECT DISTINCT cliente FROM proyectos"
                " WHERE cliente<>'' UNION SELECT DISTINCT cliente FROM documentos WHERE cliente<>''")
    con.commit()
    return con


# ---------------------------------------------------------------- reglas de revisión
def rev_valida(rev):
    """Válida: vacía, una letra (A, B, C...) o un número (0, 1, 2...)."""
    return bool(re.fullmatch(r"[A-Za-z]|\d+", rev.strip())) or rev.strip() == ""


def siguiente_rev(rev, estado=""):
    """Secuencia: A → B → C ... ; si está APROBADO pasa a 0; luego 0 → 1 → 2 ..."""
    rev = rev.strip().upper()
    if not rev:
        return "A"
    if rev.isdigit():
        return str(int(rev) + 1)
    e = _norm(estado)
    if e.startswith("aprobado") and "obs" not in e:
        return "0"
    return chr(ord(rev) + 1) if rev < "Z" else rev


def partir_estado(estado):
    """'APC (INF 7)' -> ('APC', '7')"""
    m = re.fullmatch(r"(.*?)\s*\(\s*INF\.?\s*0*(\d+)\s*\)\s*", estado or "", re.I)
    return (m.group(1).strip(), m.group(2)) if m else ((estado or "").strip(), "")


def juntar_estado(base, informe):
    base, informe = base.strip(), informe.strip()
    return f"{base} (INF {informe})" if base and informe else base


def clase_estado(estado):
    """ok / obs / mal / rev / nobs / visto / vacio — para colorear."""
    e = _norm(estado)
    if not e:
        return "vacio"
    if e.startswith("no se observa"):
        return "nobs"
    if e.startswith("visto"):
        return "visto"
    if e.startswith("aprobado c"):  # aprobado con observaciones (APROBADO C/OBS): amarillo
        return "obs"
    if e.startswith("apc") or e.startswith("aprobado"):  # APC va con Aprobado: verde
        return "ok"
    if e.startswith("dpc") or e.startswith("rechaz"):
        return "mal"
    return "rev"

# ---------------------------------------------------------------- usuarios
def _hash(clave, sal):
    return hashlib.pbkdf2_hmac("sha256", clave.encode("utf-8"), sal, 200_000).hex()


def hay_usuarios(con):
    return con.execute("SELECT COUNT(*) FROM usuarios").fetchone()[0] > 0


def usuarios(con):
    return [(r["nombre"], r["rol"]) for r in con.execute("SELECT nombre, rol FROM usuarios ORDER BY rol, nombre")]


def guardar_usuario(con, nombre, rol, clave):
    sal = os.urandom(16)
    con.execute("INSERT OR REPLACE INTO usuarios(nombre, rol, sal, clave) VALUES (?,?,?,?)",
                (nombre, rol, sal.hex(), _hash(clave, sal)))
    con.commit()


def verificar_usuario(con, nombre, clave):
    """Devuelve el rol ('admin' o 'consulta') si la clave es correcta, si no None."""
    r = con.execute("SELECT rol, sal, clave FROM usuarios WHERE nombre=?", (nombre,)).fetchone()
    if r and hmac.compare_digest(_hash(clave, bytes.fromhex(r["sal"])), r["clave"]):
        return r["rol"]
    return None

def hojas_excel(ruta):
    """Números N de las hojas 'Rev (N)' del Excel (sin abrirlo entero). None si no se puede leer."""
    import zipfile
    from html import unescape
    try:
        with zipfile.ZipFile(ruta) as z:
            xml = z.read("xl/workbook.xml").decode("utf-8", "replace")
    except Exception:  # noqa: BLE001
        return None
    nombres = [unescape(n) for n in re.findall(r'<sheet\b[^>]*?\bname="([^"]*)"', xml)]
    return sorted(int(m.group(1)) for n in nombres if (m := re.fullmatch(r"Rev \((\d+)\)", n.strip())))


def retroceder_a_excel(con, proyecto, ultima):
    """El Excel ya no tiene las hojas posteriores a Rev (ultima): se descartan esas revisiones de la base y
    se vuelve a trabajar sobre Rev (ultima). Antes se hace una copia de seguridad de la base.
    Devuelve la ruta de la copia."""
    archivo = con.execute("PRAGMA database_list").fetchone()["file"]
    carpeta = Path(archivo).parent / "respaldos"
    carpeta.mkdir(exist_ok=True)
    copia = carpeta / f"documentos antes de sincronizar {datetime.now():%Y%m%d-%H%M%S}.db"
    destino = sqlite3.connect(copia)
    con.backup(destino)
    destino.close()
    con.execute("DELETE FROM historial WHERE lista_n > ? AND documento_id IN"
                " (SELECT id FROM documentos WHERE proyecto=?)", (ultima, proyecto))
    con.execute("UPDATE proyectos SET lista_actual=? WHERE nombre=?", (ultima, proyecto))
    con.commit()
    return str(copia)


def borrar_usuarios(con):

def guardar_lista(con, proyecto, n=None, origen=None):
    """Deja el estado actual de todos los documentos como Revisión N° (la actual por defecto) en el historial.
    Se puede repetir: reemplaza lo guardado antes para esa revisión."""
    n = n or lista_actual(con, proyecto)
    for d in con.execute("SELECT id FROM documentos WHERE proyecto=?", (proyecto,)).fetchall():
        con.execute("DELETE FROM historial WHERE documento_id=? AND lista_n=?", (d["id"], n))
        _snapshot(con, d["id"], origen or f"Revisión N°{n}", lista_n=n, fecha=_hoy())
    con.commit()


def emitir_lista(con, proyecto):
    """Cierra la Revisión N° actual y abre la N+1 (cuando se entregan archivos nuevos). La N+1 arranca
    con el estado actual de los documentos."""
    n = lista_actual(con, proyecto)
    guardar_lista(con, proyecto, n)
    con.execute("UPDATE proyectos SET lista_actual=? WHERE nombre=?", (n + 1, proyecto))
    con.commit()
    guardar_lista(con, proyecto, n + 1)
    return n + 1


def resumen(con, proyecto):
    docs = buscar(con, "", proyecto)
    r = {"total": len(docs), "entregados": 0, "aprobados": 0, "en_revision": 0, "pendientes": 0}
    for d in docs:
        r["entregados"] += bool(d["entregado"])
        c = max((clase_estado(d["estado_cliente"]), clase_estado(d["estado_epen"])),
                key=["vacio", "nobs", "visto", "rev", "mal", "obs", "ok"].index)
        if c in ("ok", "obs"):
            r["aprobados"] += 1
        elif c in ("rev", "mal"):
            r["en_revision"] += 1
        else:
            r["pendientes"] += 1
    return r


# ---------------------------------------------------------------- consultas
def _filtro_texto(texto, prefijo=""):
    sql, params = "", []
    for palabra in texto.split():
        sql += (f" AND (norm({prefijo}cod_cliente) LIKE ? OR norm({prefijo}cod_epen) LIKE ?"
                f" OR norm({prefijo}titulo) LIKE ? OR norm({prefijo}estado_cliente) LIKE ?"
                f" OR norm({prefijo}estado_epen) LIKE ?)")
        params += [f"%{_norm(palabra)}%"] * 5
    return sql, params


def buscar(con, texto="", proyecto=""):
    """Documentos que entregamos nosotros (los que se controlan)."""
    sql, params = "SELECT * FROM documentos WHERE nuestro=1", []
    if proyecto:
        sql += " AND proyecto = ?"
        params.append(proyecto)
    s, p = _filtro_texto(texto)
    return con.execute(sql + s + " ORDER BY id", params + p).fetchall()


def buscar_en_lista(con, texto, proyecto, n):
    """Estado de los documentos tal como estaba en la Revisión N° de la lista."""
    s, p = _filtro_texto(texto, "d.")
    sql = ("SELECT d.id, d.proyecto, d.cliente, d.cod_cliente, d.cod_epen, d.titulo, d.responsable,"
           " d.fecha_limite_go, d.fecha_limite_cliente, h.entregado, h.rev_grupo, h.rev_cliente,"
           " h.rev_epen, h.estado_cliente, h.estado_epen, h.comentarios, h.fecha AS actualizado"
           " FROM documentos d JOIN historial h ON h.id = (SELECT MAX(id) FROM historial"
           " WHERE documento_id=d.id AND lista_n=?) WHERE d.nuestro=1 AND d.proyecto=?")
    return con.execute(sql + s + " ORDER BY d.id", [n, proyecto] + p).fetchall()


def get_doc(con, doc_id):
    return con.execute("SELECT * FROM documentos WHERE id=?", (doc_id,)).fetchone()


def historial(con, doc_id):
    return con.execute("SELECT * FROM historial WHERE documento_id=? ORDER BY id DESC",
                       (doc_id,)).fetchall()


def buscar_existente(con, proyecto, cod_epen, cod_cliente, titulo):
    for col, val in (("cod_epen", cod_epen), ("cod_cliente", cod_cliente)):
        if val:
            r = con.execute(f"SELECT id FROM documentos WHERE proyecto=? AND {col}=?",
                            (proyecto, val)).fetchone()
            if r:
                return r["id"]
    if not cod_epen and not cod_cliente and titulo:
        r = con.execute("SELECT id FROM documentos WHERE proyecto=? AND titulo=?",
                        (proyecto, titulo)).fetchone()
        if r:
            return r["id"]
    return None


# ---------------------------------------------------------------- escritura
def _snapshot(con, doc_id, origen, lista_n=None, fecha=None):
    d = get_doc(con, doc_id)
    con.execute(
        "INSERT INTO historial(documento_id,lista_n,fecha,origen,entregado,rev_grupo,rev_cliente,"
        "rev_epen,estado_cliente,estado_epen,comentarios) VALUES (?,?,?,?,?,?,?,?,?,?,?)",
        (doc_id, lista_n, fecha or _ahora(), origen, d["entregado"], d["rev_grupo"],
         d["rev_cliente"], d["rev_epen"], d["estado_cliente"], d["estado_epen"], d["comentarios"]))


def agregar(con, datos, origen="Alta", lista_n=None, fecha=None):
    """Agrega un documento. Devuelve (id, creado); si ya existe no lo duplica."""
    d = {k: (v if k in ("entregado", "nuestro") else _txt(v)) for k, v in datos.items()}
    asegurar_proyecto(con, d.get("proyecto", ""), d.get("cliente", ""))
    existente = buscar_existente(con, d.get("proyecto", ""), d.get("cod_epen", ""),
                                 d.get("cod_cliente", ""), d.get("titulo", ""))
    if existente:
        return existente, False
    cols = list(d)
    cur = con.execute(
        f"INSERT INTO documentos({','.join(cols)},actualizado) VALUES ({','.join('?' * len(cols))},?)",
        [d[c] for c in cols] + [_ahora()])
    _snapshot(con, cur.lastrowid, origen, lista_n, fecha)
    con.commit()
    return cur.lastrowid, True


def actualizar(con, doc_id, datos, origen="Edición"):
    antes = get_doc(con, doc_id)
    cols = list(datos)
    con.execute(f"UPDATE documentos SET {','.join(c + '=?' for c in cols)}, actualizado=? WHERE id=?",
                [datos[c] for c in cols] + [_ahora(), doc_id])
    if any(c in CAMPOS_ESTADO and antes[c] != datos[c] for c in cols):
        _snapshot(con, doc_id, origen)
    con.commit()


def eliminar(con, doc_id):
    con.execute("DELETE FROM documentos WHERE id=?", (doc_id,))
    con.commit()

def _n_hoja(nombre):
    m = re.search(r"\((\d+)\)", nombre)
    return int(m.group(1)) if m else None


def _es_rev(nombre):
    return re.fullmatch(r"\s*Rev\s*\((\d+)\)\s*", nombre or "") is not None


def _fecha_txt(v):
    """Fecha de una celda como dd/mm/aaaa; '' si está vacía o no es una fecha."""
    f = parse_fecha(v)
    return f.strftime("%d/%m/%Y") if f else ""


def primera_fila_kpi1(k):
    """Fila donde empiezan los documentos en la hoja «KPI N°1» (la que sigue al encabezado de la tabla):
    cambia según la versión de la hoja. Por defecto 6."""
    for r in range(1, 15):
        if _norm(k.cell(r, 1).value) == "documento cliente":
            return r + 1
    return 6


def leer_kpi1(con, wb, ws_rev, proyecto, pisar=True):
    """Fecha real de entrega (col. I) y transmittal (col. M) escritos a mano en la hoja «KPI N°1».
    La fila r de esa hoja corresponde a la fila r-2 de la hoja Rev más nueva. Con pisar=False solo
    completa lo que en la base está vacío."""
    if "KPI N°1" not in wb.sheetnames:
        return 0
    k, tocados = wb["KPI N°1"], 0
    primera = primera_fila_kpi1(k)
    for r in range(primera, primera + 100):
        a, b, c = (_txt(ws_rev.cell(r - primera + 4, i).value) for i in (1, 2, 3))  # la fila 4 de Rev va en la primera
        if not (a or b or c):
            continue
        doc = buscar_existente(con, proyecto, b, a, c)
        if not doc:
            continue
        d = get_doc(con, doc)
        for col, campo, val in (("fecha_real", "fecha_real", _fecha_txt(k.cell(r, 9).value)),
                                ("transmittal", "transmittal", _txt(k.cell(r, 13).value))):
            # al importar, el Excel manda (también si la celda está vacía); si no, solo completa huecos
            if val != d[campo] and (pisar or (val and not d[campo])):
                con.execute(f"UPDATE documentos SET {col}=? WHERE id=?", (val, doc))
                tocados += 1
    con.commit()
    return tocados


def _fecha_hoja(ws):
    m = re.search(r"\d{2}/\d{2}/\d{4}", _txt(ws["A2"].value))
    return m.group(0) if m else ""


def importar_excel(con, ruta, proyecto, cliente=""):
    """Importa el Excel de control. Cada hoja 'Rev (N)' es la Revisión N° de la lista; la más
    nueva es el estado actual. Devuelve (documentos_nuevos, filas_historial, filas_sin_coincidencia)."""
    wb = cargar_libro(ruta, data_only=True)
    hojas = sorted((h for h in wb.worksheets if _es_rev(h.title)),
                   key=lambda h: _n_hoja(h.title), reverse=True)
    if not hojas:
        raise ValueError("No encontré hojas con el formato 'Rev (N)'.")
    asegurar_proyecto(con, proyecto, cliente)

    def filas(ws):
        for r in ws.iter_rows(min_row=4, values_only=True):
            r = (list(r) + [None] * 14)[:14]
            if any(r[:3]):
                yield r

    def ciclo_de(doc, r):
        # fechas de devolución / reenvío (columnas M y N) de un documento devuelto con DPC
        if doc and "dpc" in _norm(r[10]) and _txt(r[8]) and (_txt(r[12]) or _txt(r[13])):
            guardar_ciclo(con, doc, _txt(r[8]), _txt(r[12]), _txt(r[13]), pisar=False)

    nuevos = hist = sin = 0
    actual = hojas[0]
    n_act, f_act = _n_hoja(actual.title), _fecha_hoja(actual)
    for r in filas(actual):
        fl_cli = _txt(r[4])
        entregado = 1 if fl_cli.lower() == "entregado" else 0
        doc, creado = agregar(con, dict(
            proyecto=proyecto, cliente=cliente, cod_cliente=_txt(r[0]), cod_epen=_txt(r[1]),
            titulo=_txt(r[2]), fecha_limite_go=_fecha_txt(r[3]),
            entregado=entregado, rev_grupo=_txt(r[6]), rev_cliente=_txt(r[7]), rev_epen=_txt(r[8]),
            estado_cliente=_txt(r[9]), estado_epen=_txt(r[10]), comentarios=_txt(r[11])),
            origen=f"Excel {actual.title}", lista_n=n_act, fecha=f_act)
        nuevos += creado
        if _fecha_txt(r[3]):
            con.execute("UPDATE documentos SET fecha_limite_go=? WHERE id=?", (_fecha_txt(r[3]), doc))
        ciclo_de(doc, r)
    leer_kpi1(con, wb, actual, proyecto)

    for ws in hojas[1:]:
        n, f = _n_hoja(ws.title), _fecha_hoja(ws)
        for r in filas(ws):
            doc = buscar_existente(con, proyecto, _txt(r[1]), _txt(r[0]), _txt(r[2]))
            if not doc:
                sin += 1
                continue
            ciclo_de(doc, r)
            con.execute("DELETE FROM historial WHERE documento_id=? AND lista_n=?", (doc, n))
            con.execute(
                "INSERT INTO historial(documento_id,lista_n,fecha,origen,entregado,rev_grupo,"
                "rev_cliente,rev_epen,estado_cliente,estado_epen,comentarios)"
                " VALUES (?,?,?,?,?,?,?,?,?,?,?)",
                (doc, n, f, f"Excel {ws.title}", 1 if _txt(r[4]).lower() == "entregado" else 0,
                 _txt(r[6]), _txt(r[7]), _txt(r[8]), _txt(r[9]), _txt(r[10]), _txt(r[11])))
            hist += 1
    con.execute("UPDATE proyectos SET lista_actual=MAX(lista_actual, ?), archivo_excel=? WHERE nombre=?",
                (n_act, str(ruta), proyecto))
    con.commit()
    return nuevos, hist, sin

def leer_tabla(ruta, hoja=None):
    """Lee un .xlsx/.xlsm/.csv. Devuelve (nombres_de_hojas, hoja_leida, filas)."""
    ruta = str(ruta)
    if ruta.lower().endswith((".csv", ".txt")):
        import csv
        with open(ruta, encoding="utf-8-sig", errors="replace") as f:
            txt = f.read()
        try:
            dial = csv.Sniffer().sniff(txt[:4096], delimiters=";,\t")
        except csv.Error:
            dial = csv.excel
        return ["CSV"], "CSV", [[c.strip() for c in r] for r in csv.reader(txt.splitlines(), dial)]
    wb = cargar_libro(ruta, data_only=True)
    ws = wb[hoja] if hoja in wb.sheetnames else wb.worksheets[0]
    filas = [[_txt(c) for c in r] for r in ws.iter_rows(values_only=True)]
    return wb.sheetnames, ws.title, filas


def adivinar_columnas(filas):
    """Busca la fila de encabezados y qué columna es cada dato.
    Devuelve (indice_primera_fila_de_datos, {'cod_cliente': col, 'cod_epen': col, 'titulo': col})."""
    mejor = (0, {})
    for i, r in enumerate(filas[:25]):
        m, genericas = {}, []
        for j, c in enumerate(r):
            n = _norm(c)
            if not n:
                continue
            if any(x in n for x in ("titulo", "descrip", "denominaci", "nombre")):
                m.setdefault("titulo", j)
            elif "epen" in n:
                m.setdefault("cod_epen", j)
            elif "cliente" in n or "ypf" in n:
                m.setdefault("cod_cliente", j)
            elif any(x in n for x in ("cod", "document", "numero", "nro")):
                genericas.append(j)
        for j in genericas:  # columnas de código sin nombre claro: solo si falta alguna
            for k in ("cod_epen", "cod_cliente"):
                if k not in m:
                    m[k] = j
                    break
        if len(m) > len(mejor[1]):
            mejor = (i + 1, m)
    return mejor if mejor[1] else (0, {"cod_cliente": 0, "cod_epen": 1, "titulo": 2})


def importar_listado(con, filas, primera, mapa, proyecto, cliente=""):
    """Carga el listado general de la obra como documentos que todavía NO son nuestros
    (después se marcan los que entregamos nosotros).
    mapa: {'cod_cliente': col|None, 'cod_epen': col|None, 'titulo': col|None}.
    Devuelve (nuevos, repetidos)."""
    def val(r, k):
        j = mapa.get(k)
        return _txt(r[j]) if j is not None and j < len(r) else ""
    nuevos = repes = 0
    for r in filas[primera:]:
        d = {k: val(r, k) for k in ("cod_cliente", "cod_epen", "titulo")}
        if not any(d.values()):
            continue
        _, creado = agregar(con, dict(proyecto=proyecto, cliente=cliente, nuestro=0, entregado=0, **d),
                            origen="Listado general")
        nuevos += creado
        repes += not creado
    return nuevos, repes

def _fecha_lista(con, proyecto, n):
    r = con.execute("SELECT h.fecha FROM historial h JOIN documentos d ON d.id=h.documento_id"
                    " WHERE d.proyecto=? AND h.lista_n=? ORDER BY h.id DESC LIMIT 1", (proyecto, n)).fetchone()
    m = re.search(r"\d{2}/\d{2}/\d{4}", r[0]) if r else None
    return m.group(0) if m else _hoy()


def _rev_xl(v):
    return int(v) if v.isdigit() else v


def _dt(v):
    """Texto dd/mm/aaaa -> fecha para escribir en una celda de Excel (None si está vacío)."""
    f = parse_fecha(v)
    return datetime(f.year, f.month, f.day) if f else None


def _hoja_nueva(wb, nombre, proyecto, cliente, n, filas, fecha, con=None):
    from openpyxl.styles import Font, PatternFill, Alignment, Border, Side
    borde = Side(style="thin", color="999999")
    ws = wb.create_sheet(nombre)
    ws.merge_cells("A1:N1")
    ws.merge_cells("A2:N2")
    ws["A1"] = proyecto
    ws["A2"] = f"Revisión N°{n} - Ultima Actualización: {fecha}"
    for c in ("A1", "A2"):
        ws[c].font = Font(bold=True, size=14 if c == "A1" else 11)
        ws[c].alignment = Alignment(horizontal="center")
    c = (cliente or "EPEN").strip().upper()
    ws.append(["DOCUMENTO CLIENTE", f"DOCUMENTO {c}",
               "TITULO DEL DOCUMENTO", "FECHA LIMITE GRUPO OESTE (CRONOGRAMA)", "ESTADO",
               "RESPONSABLE", "ESTADO GRUPO OESTE", "REVISION EMITIDA AL CLIENTE",
               f"REVISION EMITIDA A {c}", "ESTADO CLIENTE", f"ESTADO {c}", "COMENTARIOS",
               "FECHA DEVOLUCIÓN", "FECHA REENVÍO"])
    for c in ws[3]:
        c.font = Font(bold=True, color="FFFFFF")
        c.fill = PatternFill("solid", fgColor="1F4E78")
        c.alignment = Alignment(horizontal="center", vertical="center", wrap_text=True)
    for d in filas:
        dev = rei = None
        if con is not None and "dpc" in _norm(d["estado_epen"]) and d["rev_epen"]:
            cy = ciclo(con, d["id"], d["rev_epen"])
            if cy:
                dev, rei = _dt(cy["fecha_devolucion"]), _dt(cy["fecha_reenvio"])
        ws.append([d["cod_cliente"], d["cod_epen"], d["titulo"], _dt(d["fecha_limite_go"]),
                   "Entregado" if d["entregado"] else "", "",
                   _rev_xl(d["rev_grupo"]), _rev_xl(d["rev_cliente"]), _rev_xl(d["rev_epen"]),
                   d["estado_cliente"], d["estado_epen"], d["comentarios"], dev, rei])
    for row in ws.iter_rows(min_row=3, max_row=ws.max_row):
        for c in row:
            c.border = Border(left=borde, right=borde, top=borde, bottom=borde)
            if c.row > 3:
                c.alignment = Alignment(vertical="center", wrap_text=True,
                                        horizontal="center" if c.column > 3 else None)
    for i, d in enumerate(filas):  # mismo criterio de colores que las hojas que se agregan al Excel vinculado
        fila = 4 + i
        if _color_fila(d):
            for col in range(1, 13):
                ws.cell(fila, col).fill = PatternFill("solid", fgColor="FF92D050")
        if clase_estado(d["estado_epen"] or d["estado_cliente"]) == "mal":
            ws.cell(fila, 11).fill = PatternFill("solid", fgColor="FFFFFF00")
    for col, w in zip("ABCDEFGHIJKLMN", [36, 30, 70, 20, 16, 16, 12, 14, 14, 18, 28, 30, 16, 16]):
        ws.column_dimensions[col].width = w
    for row in ws.iter_rows(min_row=4, max_row=ws.max_row, min_col=4, max_col=4):
        row[0].number_format = "dd/mm/yyyy"
    for col in (13, 14):
        for r in range(4, ws.max_row + 1):
            ws.cell(r, col).number_format = "dd/mm/yyyy"
    ws.row_dimensions[3].height = 45
    ws.freeze_panes = "A4"



def _color_fila(d):
    """Criterio de colores del Excel original: fila verde = entregado y aprobado (revisión numérica
    o estado APROBADO); amarillo (solo en ESTADO EPEN) = DPC / rechazado; sin color = resto."""
    est = d["estado_epen"] or d["estado_cliente"]
    if d["entregado"] and (d["rev_grupo"].isdigit() or clase_estado(est) == "ok"):
        return "FF92D050"
    return None


def _preparar_hoja_rev(ws, cliente):
    """Deja la hoja con el formato actual: ESTADO (antes «FECHA LIMITE CLIENTE»), fecha límite del cronograma y
    las columnas FECHA DEVOLUCIÓN / FECHA REENVÍO (M y N) que usa el KPI de respuesta a correcciones."""
    from copy import copy
    ws.cell(3, 4).value = "FECHA LIMITE GRUPO OESTE (CRONOGRAMA)"
    ws.cell(3, 5).value = "ESTADO"
    for col, titulo in ((13, "FECHA DEVOLUCIÓN"), (14, "FECHA REENVÍO")):
        c = ws.cell(3, col)
        if c.value is None:
            c._style = copy(ws.cell(3, 12)._style)
            c.value = titulo
        ws.column_dimensions[c.column_letter].width = max(ws.column_dimensions[c.column_letter].width or 0, 16)
        for r in range(4, ws.max_row + 1):
            ws.cell(r, col).number_format = "dd/mm/yyyy"
    ws.column_dimensions["D"].width = max(ws.column_dimensions["D"].width or 0, 20)
    for r in range(4, ws.max_row + 1):
        if ws.cell(r, 4).value is not None and not isinstance(ws.cell(r, 4).value, str):
            ws.cell(r, 4).number_format = "dd/mm/yyyy"


def agregar_hoja_excel(con, ruta, proyecto, n, carpeta_respaldo=None):
    """Agrega al Excel de control la hoja 'Rev (n)' con el estado actual de los documentos y regenera las hojas
    «KPI N°1», «KPI N°2» y «Dashboard». Copia el formato de la última hoja 'Rev (N)' del archivo. Antes hace una
    copia de respaldo. Devuelve la ruta del respaldo."""
    import shutil
    from copy import copy
    from openpyxl.styles import PatternFill
    import kpi

    ruta = Path(ruta)
    wb = cargar_libro(ruta, keep_vba=ruta.suffix.lower() == ".xlsm")
    nombre = f"Rev ({n})"
    if nombre in wb.sheetnames:
        raise ValueError(f"El Excel ya tiene una hoja «{nombre}». No la pisé para no perder datos.")
    carpeta = Path(carpeta_respaldo or ruta.parent / "respaldos")
    carpeta.mkdir(exist_ok=True)
    respaldo = carpeta / f"{ruta.stem} - antes de Rev ({n}) {datetime.now():%Y%m%d-%H%M%S}{ruta.suffix}"
    shutil.copy2(ruta, respaldo)
    kpi.leer_entradas(wb, con, proyecto)  # lo que se escribió a mano en el Excel completa la base
    docs = buscar(con, "", proyecto)
    previas = sorted((h for h in wb.worksheets if _es_rev(h.title)), key=lambda h: _n_hoja(h.title))
    if not previas:
        _hoja_nueva(wb, nombre, proyecto, cliente_de(con, proyecto), n, docs, _hoy(), con)
        kpi.construir(wb, con, proyecto)
        kpi.activar(wb, nombre)
        wb.save(ruta)
        return str(respaldo)

    src = previas[-1]
    ws = wb.copy_worksheet(src)
    ws.title = nombre
    pos_rev = [i for i, h in enumerate(wb.worksheets) if h is not ws and _es_rev(h.title)]
    # orden «más nueva primero»: la nueva va antes de la última Rev; si no, al final de las Rev
    destino = pos_rev[0] if wb.worksheets[pos_rev[0]] is src else pos_rev[-1] + 1
    wb.move_sheet(ws, offset=destino - (len(wb.sheetnames) - 1))
    ws["A2"] = f"Revisión N°{n} - Ultima Actualización: {_hoy()}"
    ws["A1"] = proyecto
    c = (cliente_de(con, proyecto) or "EPEN").strip().upper()  # el nombre del cliente reemplaza a «EPEN»
    ws.cell(3, 2).value = f"DOCUMENTO {c}"
    ws.cell(3, 9).value = f"REVISION EMITIDA A {c}"
    ws.cell(3, 11).value = f"ESTADO {c}"
    _preparar_hoja_rev(ws, c)

    _escribir_docs(ws, con, docs)
    kpi.construir(wb, con, proyecto)
    kpi.activar(wb, nombre)
    wb.save(ruta)
    return str(respaldo)


def _escribir_docs(ws, con, docs):
    """Vuelca el estado de los documentos en una hoja Rev (la busca por código; agrega al final los que faltan)."""
    from copy import copy
    from openpyxl.styles import PatternFill
    por_epen = {d["cod_epen"]: d for d in docs if d["cod_epen"]}
    por_cli = {d["cod_cliente"]: d for d in docs if d["cod_cliente"]}
    usados, ultima = set(), 3
    vacio = copy(ws["G4"].fill)
    verde = PatternFill("solid", fgColor="FF92D050")
    amarillo = PatternFill("solid", fgColor="FFFFFF00")

    def escribir(fila, d):
        ws.cell(fila, 5).value = "Entregado" if d["entregado"] else None
        if parse_fecha(d["fecha_limite_go"]):
            ws.cell(fila, 4).value = _dt(d["fecha_limite_go"])
            ws.cell(fila, 4).number_format = "dd/mm/yyyy"
        for col, k in ((7, "rev_grupo"), (8, "rev_cliente"), (9, "rev_epen")):
            ws.cell(fila, col).value = _rev_xl(d[k]) if d[k] != "" else None
        ws.cell(fila, 10).value = d["estado_cliente"] or None
        ws.cell(fila, 11).value = d["estado_epen"] or None
        ws.cell(fila, 12).value = d["comentarios"] or None
        cy = ciclo(con, d["id"], d["rev_epen"]) if "dpc" in _norm(d["estado_epen"]) and d["rev_epen"] else None
        ws.cell(fila, 13).value = _dt(cy["fecha_devolucion"]) if cy else None
        ws.cell(fila, 14).value = _dt(cy["fecha_reenvio"]) if cy else None
        color = _color_fila(d)
        for col in range(1, 13):  # la fila completa, de DOCUMENTO CLIENTE a COMENTARIOS
            ws.cell(fila, col).fill = verde if color else copy(vacio)
        if clase_estado(d["estado_epen"] or d["estado_cliente"]) == "mal":
            ws.cell(fila, 11).fill = amarillo

    for fila in range(4, ws.max_row + 1):
        a, b, c = _txt(ws.cell(fila, 1).value), _txt(ws.cell(fila, 2).value), _txt(ws.cell(fila, 3).value)
        if not (a or b or c):
            continue
        ultima = fila
        d = por_epen.get(b) if b else None
        d = d or (por_cli.get(a) if a else None)
        if d:
            usados.add(d["id"])
            escribir(fila, d)

    plantilla = ultima
    for d in docs:  # documentos que no estaban en la hoja anterior
        if d["id"] in usados:
            continue
        ultima += 1
        for col in range(1, 15):
            ws.cell(ultima, col)._style = copy(ws.cell(plantilla, col)._style)
        ws.cell(ultima, 1).value = d["cod_cliente"] or None
        ws.cell(ultima, 2).value = d["cod_epen"] or None
        ws.cell(ultima, 3).value = d["titulo"] or None
        ws.cell(ultima, 4).value = None
        escribir(ultima, d)

def exportar_excel(con, ruta, proyecto):
    """Una hoja 'Rev (N)' por cada revisión de la lista, con las mismas columnas del Excel original,
    más las hojas «KPI N°1», «KPI N°2» y «Dashboard»."""
    from openpyxl import Workbook
    import kpi
    cliente = cliente_de(con, proyecto)
    actual = lista_actual(con, proyecto)
    wb = Workbook()
    wb.remove(wb.active)
    for n in range(actual, 0, -1):
        filas = buscar(con, "", proyecto) if n == actual else buscar_en_lista(con, "", proyecto, n)
        if not filas and n != actual:
            continue
        _hoja_nueva(wb, f"Rev ({n})", proyecto, cliente, n, filas,

    kpi.construir(wb, con, proyecto)

    wb.save(ruta)

"""KPI del control de documentos.

* KPI N°2 · respuesta a correcciones: días entre que el documento vuelve con DPC (FECHA DEVOLUCIÓN) y se
  reenvía corregido (FECHA REENVÍO). Cada ciclo es un documento + la revisión devuelta.
* Dashboard gerencial: tarjetas, estados, plazos y los documentos que requieren atención.

Las hojas de Excel se generan con fórmulas (igual que las del Excel original), así siguen funcionando si se
cargan fechas a mano. Las funciones kpi1 / kpi2 calculan lo mismo en Python para mostrarlo dentro de la app.
"""

# ============================================================ cálculo en Python (para la app y las pruebas)
def evaluar(d, hoy=None):
    """Fecha planificada contra fecha real de entrega de un documento (mismas reglas que la hoja KPI N°1)."""
    hoy = hoy or date.today()
    plan, real = db.parse_fecha(d["fecha_limite_go"]), db.parse_fecha(d["fecha_real"])
    desvio = (real - plan).days if plan and real else None
    cumpl = "Sin datos" if desvio is None else "En término" if desvio <= 0 else "Atrasado"
    if plan is None:
        plazo = "Sin planificación"
    elif real:
        plazo = "Entregado en término" if real <= plan else "Entregado con atraso"
    elif plan < hoy:
        plazo = "Vencido"
    elif (plan - hoy).days <= 7:
        plazo = "Próximo a vencer"
    else:
        plazo = "Pendiente en plazo"
    return dict(plan=plan, real=real, desvio=desvio, cumplimiento=cumpl, plazo=plazo)


def kpi1(con, proyecto, hoy=None):
    """Cumplimiento de entrega de cada documento que entregamos. Devuelve (filas, resumen)."""
    filas = []
    for d in db.buscar(con, "", proyecto):
        filas.append(dict(doc=d, transmittal=d["transmittal"], **evaluar(d, hoy)))
    en_t = sum(f["cumplimiento"] == "En término" for f in filas)
    atr = sum(f["cumplimiento"] == "Atrasado" for f in filas)
    desvios = [f["desvio"] for f in filas if f["desvio"] is not None]
    atraso = [f for f in filas if f["desvio"] is not None and f["desvio"] > 0]
    peor = max(atraso, key=lambda f: f["desvio"]) if atraso else None
    res = dict(max_desvio=peor["desvio"] if peor else None,
               max_doc=f"{peor['doc']['cod_epen']} - {peor['doc']['titulo']}" if peor else "",
               prom_desvio=(sum(desvios) / len(desvios)) if desvios else None,
               total=len(filas), entregados=sum(bool(f["doc"]["entregado"]) for f in filas),
               en_termino=en_t, atrasados=atr, evaluadas=en_t + atr,
               cumplimiento=(en_t / (en_t + atr)) if en_t + atr else None,
               vencidos=sum(f["plazo"] == "Vencido" for f in filas),
               plazos={p: sum(f["plazo"] == p for f in filas) for p in PLAZOS})
    return filas, res


def _siguiente_letra(rev):
    rev = (rev or "").strip().upper()
    return chr(ord(rev) + 1) if len(rev) == 1 and "A" <= rev < "Z" else ""


def kpi2(con, proyecto, hoy=None, hasta=None):
    """Ciclos de corrección: documentos que volvieron con DPC. Cada uno cuenta una sola vez (documento + revisión
    devuelta). Devuelve (ciclos, resumen)."""
    hoy = hoy or date.today()
    actual = db.lista_actual(con, proyecto)
    vistos, ciclos = set(), []
    for n in range(1, (hasta or actual) + 1):
        filas = db.buscar(con, "", proyecto) if n == actual else db.buscar_en_lista(con, "", proyecto, n)
        for d in filas:
            if not d["cod_epen"] or "dpc" not in db._norm(d["estado_epen"]) or (d["id"], d["rev_epen"]) in vistos:
                continue
            vistos.add((d["id"], d["rev_epen"]))
            cy = db.ciclo(con, d["id"], d["rev_epen"])
            dev = db.parse_fecha(cy["fecha_devolucion"]) if cy else None
            rei = db.parse_fecha(cy["fecha_reenvio"]) if cy else None
            ciclos.append(dict(doc_id=d["id"], cod_epen=d["cod_epen"], cod_cliente=d["cod_cliente"], titulo=d["titulo"],
                               rev_devuelta=d["rev_epen"], nueva_rev=_siguiente_letra(d["rev_epen"]),
                               devolucion=dev, reenvio=rei, lista=n,
                               dias=(rei - dev).days if dev and rei else None,
                               estado="Cerrado" if rei else "Pendiente",
                               acum=((rei or hoy) - dev).days if dev else None))
    por_doc = {}
    for c in ciclos:
        por_doc[c["cod_epen"]] = por_doc.get(c["cod_epen"], 0) + (c["acum"] or 0)
    for c in ciclos:
        c["acum_doc"] = por_doc[c["cod_epen"]]
    dias = [c["dias"] for c in ciclos if c["dias"] is not None]
    con_fecha = [c for c in ciclos if c["acum"] is not None]
    peor = max(ciclos, key=lambda c: c["acum_doc"]) if con_fecha else None
    res = dict(promedio=(sum(dias) / len(dias)) if dias else None, ciclos=len(ciclos),
               cerrados=sum(c["estado"] == "Cerrado" for c in ciclos),
               pendientes=sum(c["estado"] == "Pendiente" for c in ciclos),
               mayor_demora=peor["acum_doc"] if peor else None,
               mayor_doc=f"{peor['cod_epen']} - {peor['titulo']}" if peor else "")
    return ciclos, res


_CACHE_RESUMEN = {}


def resumen_excel(ruta):
    """Máximo desvío, documento y promedio de desvío tal como los calculó Excel en la fila 3 de la hoja «KPI N°1»
    (valores guardados al cerrar el archivo). None si no se puede leer o la hoja no tiene valores calculados."""
    import os
    try:
        marca = (str(ruta), os.path.getmtime(ruta))
    except OSError:
        return None
    if marca in _CACHE_RESUMEN:
        return _CACHE_RESUMEN[marca]
    res = None
    try:
        wb = db.cargar_libro(ruta, data_only=True, read_only=True)
        if KPI1 in wb.sheetnames:
            ws = wb[KPI1]
            fila = {}
            for celdas in ws.iter_rows(min_row=3, max_row=3):
                for c in celdas:
                    if c.value is not None:
                        fila[c.column] = c.value
            # B3 = máximo desvío, D3 = documento, L3 = promedio (misma disposición que el Excel)
            mx, doc, prom = fila.get(2), fila.get(4), fila.get(12)
            if mx is not None or prom is not None:
                res = dict(max_desvio=mx if isinstance(mx, (int, float)) else None, max_doc=str(doc or ""),
                           prom_desvio=prom if isinstance(prom, (int, float)) else None)
        wb.close()
    except Exception:  # noqa: BLE001
        res = None
    _CACHE_RESUMEN[marca] = res
    return res


# ============================================================ lo que se escribe a mano en el Excel -> base
def _revs(wb):
    return sorted(((db._n_hoja(h.title), h) for h in wb.worksheets if db._es_rev(h.title)), key=lambda t: t[0])


def _codigos(ws, f):
    return tuple(db._txt(ws.cell(f, i).value) for i in (1, 2, 3))


def leer_entradas(wb, con, proyecto):
    """Trae a la base lo que se cargó a mano en el Excel desde la última vez: fechas de devolución / reenvío
    (columnas M y N de las hojas Rev) y fecha real / transmittal (hoja KPI N°1). No pisa lo que ya tiene la base."""
    revs = _revs(wb)
    for _, ws in revs:
        for f in range(4, ws.max_row + 1):
            a, b, c = _codigos(ws, f)
            rev = db._txt(ws.cell(f, 9).value)
            if not (a or b or c) or not rev or "dpc" not in db._norm(ws.cell(f, 11).value):
                continue
            dev, rei = db._fecha_txt(ws.cell(f, 13).value), db._fecha_txt(ws.cell(f, 14).value)
            doc = db.buscar_existente(con, proyecto, b, a, c) if (dev or rei) else None
            if doc:
                db.guardar_ciclo(con, doc, rev, dev, rei, pisar=False)
    if revs:
        db.leer_kpi1(con, wb, revs[-1][1], proyecto, pisar=False)


def sincronizar_ciclos(wb, con, proyecto):
    """Escribe en las columnas M y N de todas las hojas Rev las fechas de devolución / reenvío de cada documento
    que está en DPC (el KPI N°2 las lee de ahí). Nunca borra lo que ya está escrito."""
    for _, ws in _revs(wb):
        for col, titulo in ((13, "FECHA DEVOLUCIÓN"), (14, "FECHA REENVÍO")):
            c = ws.cell(3, col)
            if c.value is None:
                c._style = copy(ws.cell(3, 12)._style)
                c.value = titulo
            ws.column_dimensions[c.column_letter].width = max(ws.column_dimensions[c.column_letter].width or 0, 16)
        for f in range(4, ws.max_row + 1):
            a, b, c = _codigos(ws, f)
            rev = db._txt(ws.cell(f, 9).value)
            if not (a or b or c) or not rev or "dpc" not in db._norm(ws.cell(f, 11).value):
                continue
            doc = db.buscar_existente(con, proyecto, b, a, c)
            cy = db.ciclo(con, doc, rev) if doc else None
            for col, campo in ((13, "fecha_devolucion"), (14, "fecha_reenvio")):
                v = db._dt(cy[campo]) if cy else None
                if v:
                    cel = ws.cell(f, col)
                    if not cel.has_style:
                        cel._style = copy(ws.cell(f, 12)._style)
                    cel.value, cel.number_format = v, "dd/mm/yyyy"


# ============================================================ hojas de Excel
def _borde(color):
    s = Side(style="thin", color=color)
    return Border(left=s, right=s, top=s, bottom=s)


def _celda(ws, ref, valor=None, *, tam=9, neg=False, color="1F1F1F", fondo=None, h=None, v="center",
           ajuste=False, nf=None, borde=None, fuente="Calibri"):
    c = ws[ref]
    if valor is not None:
        c.value = valor
    c.font = Font(name=fuente, size=tam, bold=neg, color=color)
    if fondo:
        c.fill = PatternFill("solid", fgColor=fondo)
    c.alignment = Alignment(horizontal=h, vertical=v, wrap_text=ajuste)
    if nf:
        c.number_format = nf
    if borde:
        c.border = _borde(borde)
    return c


def _regla(formula, fondo, texto, neg=False):
    return FormulaRule(formula=[formula], fill=PatternFill(start_color=fondo, end_color=fondo, fill_type="solid"),
                       font=Font(color=texto, bold=neg))


def _ref(col, f):
    """Celda de la hoja Rev elegida en B2, con el patrón que usa el Excel original."""
    r = f'INDIRECT("\'"&$B$2&"\'!{col}{f}")'
    return f'=IFERROR(IF(COUNTA({r})=0,"",{r}),"")'

def activar(wb, nombre):
    """Deja una sola hoja seleccionada (si quedan varias «agrupadas», lo que se escribe en una se copia en todas)."""
    wb.active = wb.sheetnames.index(nombre)
    for h in wb.worksheets:
        h.sheet_view.tabSelected = h.title == nombre


def construir(wb, con, proyecto):
    """Regenera «KPI N°1», «KPI N°2» y «Dashboard» (al final del libro, después de las hojas Rev).
    Devuelve False si el libro no tiene hojas Rev."""
    revs = _revs(wb)
    if not revs:
        return False
    sincronizar_ciclos(wb, con, proyecto)
    for nombre in (KPI1, KPI2, DASH):
        if nombre in wb.sheetnames:
            del wb[nombre]
    n_ult, ult = revs[-1]
    filas_doc = {}
    for f in range(4, 4 + FILAS_KPI1):
        a, b, c = _codigos(ult, f)
        if a or b or c:
            doc = db.buscar_existente(con, proyecto, b, a, c)
            if doc:
                filas_doc[f] = db.get_doc(con, doc)
    otras = [h for h in wb.worksheets if not db._es_rev(h.title) and h.title not in (KPI1, KPI2, DASH)]
    fuente = ('="Rev ("&(_xlfn.SHEETS()-3+0*NOW())&")"' if not otras else f'="Rev ({n_ult})"')
    ws1, ws2, ws3 = wb.create_sheet(KPI1), wb.create_sheet(KPI2), wb.create_sheet(DASH)
    _hoja_kpi1(ws1, filas_doc, fuente)
    filas_por_hoja = max(32, max(h.max_row for _, h in revs) - 2)
    _hoja_kpi2(ws2, max(20, n_ult + 5), filas_por_hoja)
    _hoja_dashboard(ws3)
    wb.calculation.fullCalcOnLoad = True
    return True

"""Lectura de los «Informes de ingeniería» en PDF: toma la primera tabla (Nº, Rev, Título, Observaciones)
y la convierte en estados para actualizar el control de documentos."""
import re
import unicodedata
from pathlib import Path

import db


def _fitz():
    try:
        import pymupdf as fz
    except ImportError:
        try:
            import fitz as fz
        except ImportError as e:  # pragma: no cover
            raise RuntimeError("Falta la librería para leer PDF. Instalala con:  pip install pymupdf") from e
    return fz


def clave(codigo):
    """Código sin espacios ni guiones, para comparar «ETLC -0226 -ET -Cm…» con «ETLC-0226-ET-Cm…»."""
    return re.sub(r"[\s\-–—_.]", "", codigo or "").upper()


def _compacto(t):
    t = unicodedata.normalize("NFD", (t or "").lower())
    return re.sub(r"[^a-z0-9]", "", "".join(c for c in t if unicodedata.category(c) != "Mn"))


def clasificar_obs(texto):
    """'Aprobado con observaciones' -> 'APROBADO C/OBS'; 'APC' -> 'APC'; 'Devuelto para corrección' -> 'DPC'; 'Aprobado' -> 'APROBADO'.
    Devuelve (estado, reconocido)."""
    c = _compacto(texto)
    if c.startswith("aprobadocon"):
        return "APROBADO C/OBS", True
    if c.startswith("apc"):
        return "APC", True
    if c.startswith(("devuelto", "dpc")):
        return "DPC", True
    if c.startswith("rechaz"):
        return "RECHAZADO", True
    if c.startswith("aprobado"):
        return "APROBADO", True
    if c.startswith("enrevision"):
        return "EN REVISION", True
    return re.sub(r"\s+", " ", (texto or "").strip()), False


def leer_informe(ruta):
    """Devuelve {'numero': '8', 'fecha': '05/10/2026', 'envio': '15', 'filas': [{'codigo','rev','titulo','obs'}]}."""
    fz = _fitz()
    doc = fz.open(str(ruta))
    texto1 = doc[0].get_text() if len(doc) else ""
    m = re.search(r"INFORME\s+DE\s+INGENIER[ÍI]A\s+N[º°o\.]*\s*0*(\d+)", texto1, re.I)
    numero = m.group(1) if m else ""
    if not numero:
        m = re.search(r"informe[\s\-_]*0*(\d+)", Path(ruta).stem, re.I)
        numero = m.group(1) if m else ""
    fecha = (re.search(r"\d{2}/\d{2}/\d{4}", texto1) or [""])[0] if texto1 else ""
    envio = (re.search(r"ENVIO\s+N[º°o\.]*\s*(\d+)", texto1, re.I) or [None, ""])[1]

    filas, hubo_encabezado = [], False
    for pagina in doc:
        for t in pagina.find_tables().tables:
            datos = t.extract()
            if not datos or len(datos[0]) < 3:
                continue
            cab = [_compacto(c) for c in datos[0]]
            if any("rev" == c for c in cab) and any(c.startswith("observ") for c in cab):
                hubo_encabezado = True
                i_cod = 0
                i_rev = next(i for i, c in enumerate(cab) if c == "rev")
                i_obs = next(i for i, c in enumerate(cab) if c.startswith("observ"))
                i_tit = next((i for i, c in enumerate(cab) if c.startswith("titulo")), 2)
                cuerpo = datos[1:]
            elif hubo_encabezado and len(datos[0]) == 4 and re.fullmatch(r"[A-Za-z0-9]{1,2}", (datos[0][1] or "").strip()):
                i_cod, i_rev, i_tit, i_obs = 0, 1, 2, 3  # la tabla continúa en otra página
                cuerpo = datos
            else:
                continue
            for r in cuerpo:
                r = [(c or "").strip() for c in r]
                if len(r) <= max(i_cod, i_rev, i_obs) or not clave(r[i_cod]):
                    continue
                filas.append({"codigo": re.sub(r"\s+", " ", r[i_cod]), "rev": r[i_rev].upper(),
                              "titulo": re.sub(r"\s+", " ", r[i_tit]) if i_tit < len(r) else "",
                              "obs": re.sub(r"\s+", " ", r[i_obs])})
        if hubo_encabezado and filas:
            pass  # sigue buscando en las demás páginas solo continuaciones de la misma tabla
    return {"numero": numero, "fecha": fecha, "envio": envio, "filas": filas}


def _inf(estado):
    n = db.partir_estado(estado)[1]
    return int(n) if n.isdigit() else 0


def ultimo_informe(con, proyecto):
    """Número del último informe ya aplicado: el mayor «(INF N)» que figura en los estados de los documentos."""
    return max([_inf(d["estado_epen"]) for d in db.buscar(con, "", proyecto)] +
               [_inf(d["estado_cliente"]) for d in db.buscar(con, "", proyecto)] + [0])


def informe_viejo(con, proyecto, informe):
    """(es_viejo, ultimo_aplicado): el informe cargado es anterior al último informe ya aplicado."""
    ult = ultimo_informe(con, proyecto)
    return (informe["numero"].isdigit() and int(informe["numero"]) < ult), ult


def emparejar(con, proyecto, informe, campo="estado_epen"):
    """Une cada fila del informe con un documento del control (por código EPEN o del cliente).
    Devuelve una lista de dicts: fila, doc (o None), estado (nuevo), reconocido, aviso, viejo, igual.
    Si el informe es anterior al último ya aplicado, todas las filas encontradas quedan «viejo» (no se pueden aplicar)."""
    docs = db.buscar(con, "", proyecto)
    es_viejo, ult = informe_viejo(con, proyecto, informe)
    por_clave = {}
    for d in docs:
        for k in (d["cod_epen"], d["cod_cliente"]):
            if clave(k):
                por_clave.setdefault(clave(k), d)
    res = []
    for f in informe["filas"]:
        d = por_clave.get(clave(f["codigo"]))
        base, ok = clasificar_obs(f["obs"])
        nuevo = db.juntar_estado(base, informe["numero"]) if ok else base
        aviso, viejo, igual = "", False, False
        if d is None:
            aviso = "No está en la lista de documentos del control"
        elif es_viejo:
            viejo = True
            sig = max(_inf(d["estado_epen"]), _inf(d["estado_cliente"]))
            n_inf = int(informe["numero"])
            donde = (f"el documento ya pasó al informe N°{sig}" if sig > n_inf
                     else f"el último informe aplicado es el N°{ult}")
            aviso = f"Informe viejo: {donde} (este es el N°{n_inf})"
        else:
            rev_inf = (f["rev"] or "").strip().upper()
            revs_ok = (not rev_inf) or all(d[k].upper() == rev_inf for k in ("rev_grupo", "rev_cliente", "rev_epen"))
            if rev_inf and not revs_ok:
                aviso = f"Revisión: {d['rev_epen'] or d['rev_grupo'] or '—'} → {rev_inf}"
            if d[campo].strip() == nuevo and revs_ok and d["entregado"]:
                igual, aviso = True, "Sin cambios: ya tiene este estado y esta revisión"
        if not ok and not viejo:
            aviso = (aviso + " · " if aviso else "") + "Observación no reconocida: se guardaría tal cual"
        res.append({"fila": f, "doc": d, "estado": nuevo, "reconocido": ok, "aviso": aviso, "viejo": viejo, "igual": igual})
    return res

        enc = self.filas[primera - 1] if primera >= 1 and self.filas else []

        self.destroy()

    def _llenar(self):
        self.filas = informe.emparejar(self.con, self.proyecto, self.info, self.v_campo.get())
        self.marca = [r["doc"] is not None and not r["viejo"] and not r["igual"] for r in self.filas]
        self.tree.delete(*self.tree.get_children())
        for i, r in enumerate(self.filas):
            d = r["doc"]
            tag = ("mal",) if d is None else ("igual",) if (r["igual"] or r["viejo"]) else ("aviso",) if r["aviso"] else ()
            self.tree.insert("", "end", iid=str(i), tags=tag, values=(
                "🔒" if r["viejo"] else "=" if r["igual"] else "☑" if self.marca[i] else "☐" if d else "—",
                (d["cod_epen"] or d["cod_cliente"]) if d else r["fila"]["codigo"].replace(" ", ""),
                d["titulo"] if d else r["fila"]["titulo"], r["fila"]["rev"],
                d[self.v_campo.get()] if d else "", r["estado"], r["aviso"]))
        self._contar()

    def _contar(self):
        iguales = sum(1 for r in self.filas if r["igual"])
        viejos = sum(1 for r in self.filas if r["viejo"])
        nuevos = sum(1 for r in self.filas if r["doc"] is None)
        self.lbl.config(text=f"{sum(self.marca)} se van a actualizar · {iguales} ya estaban al día · "
                             f"{nuevos} no encontrados" + (f" · {viejos} bloqueados (informe viejo)" if viejos else ""))
        self.btn.state(["!disabled"] if any(self.marca) else ["disabled"])

    def _clic(self, ev):
        if self.tree.identify_region(ev.x, ev.y) != "cell":
            return
        iid = self.tree.identify_row(ev.y)
        if not iid or self.filas[int(iid)]["doc"] is None or self.filas[int(iid)]["igual"] or self.filas[int(iid)]["viejo"]:
            return
        i = int(iid)
        self.marca[i] = not self.marca[i]
        self.tree.set(iid, "ok", "☑" if self.marca[i] else "☐")
        self._contar()

    def _aplicar(self):
        campo = self.v_campo.get()
        origen = f"Informe {self.info['numero']} (PDF)"
        for r, on in zip(self.filas, self.marca):
            if on and r["doc"] is not None and not r["igual"] and not r["viejo"]:
                datos = {campo: r["estado"], "entregado": 1}  # si está en un informe, se entregó
                rev = (r["fila"]["rev"] or "").strip().upper()
                if rev and db.rev_valida(rev):  # la revisión del informe (A, B, 0, 1…) va a las tres columnas
                    datos.update(rev_grupo=rev, rev_cliente=rev, rev_epen=rev)
                db.actualizar(self.con, r["doc"]["id"], datos, origen=origen)
                self.aplicado += 1
        self.destroy()

    def _visibles(self):
        palabras = [db._norm(p) for p in self.v_q.get().split()]
        for d in self.docs:
            txt = db._norm(f"{d['cod_epen']} {d['cod_cliente']} {d['titulo']}")
            if all(p in txt for p in palabras):
                yield d

    def _llenar(self):
        self.tree.delete(*self.tree.get_children())
        for d in self._visibles():
            self._fila(d)
        self._contar()

    def _fila(self, d, **_):
        on = self.marca[d["id"]]
        self.tree.insert("", "end", iid=str(d["id"]), tags=("si",) if on else (),
                         values=("☑" if on else "☐", d["cod_epen"], d["cod_cliente"], d["titulo"]))

        self.lbl.config(text=f"{sum(self.marca.values())} marcados de {len(self.docs)} documentos")

        if not iid:
            return
        i = int(iid)
        self.marca[i] = not self.marca[i]
        on = self.marca[i]
        self.tree.set(iid, "ok", "☑" if on else "☐")
        self.tree.item(iid, tags=("si",) if on else ())
        self._contar()

    def _clic(self, ev):
        if self.tree.identify_region(ev.x, ev.y) == "cell":
            self._alternar(self.tree.identify_row(ev.y))

    def _todos(self, valor):
        for d in self._visibles():
            self.marca[d["id"]] = valor
        self._llenar()

    def _guardar(self):
        db.marcar_nuestros(self.con, [i for i, v in self.marca.items() if v],
                           [i for i, v in self.marca.items() if not v])
        self.destroy()


def _centrar(win, ancho, alto):
    win.update_idletasks()
    x = (win.winfo_screenwidth() - ancho) // 2
    y = max((win.winfo_screenheight() - alto) // 3, 0)
    win.geometry(f"{ancho}x{alto}+{x}+{y}")

class ContextoDialog(_Dialogo):
    """Después de ingresar: elegir (o escribir) el proyecto y el cliente con los que se va a trabajar."""

    def __init__(self, master, con, rol):
        super().__init__(master, "Proyecto y cliente", 500, 380)
        self.con, self.rol, self.resultado = con, rol, None
        admin = rol == "admin"
        ttk.Label(self, text="¿Con qué proyecto trabajás?", style="Big.TLabel").pack(anchor="w", padx=22, pady=(18, 2))
        ttk.Label(self, style="Hint.TLabel", justify="left", wraplength=455, text=(
            "Proyecto: elegí uno de los que ya usaste o «➕ Nuevo proyecto…» (lo escribís vos). "
            "Cliente: «EPEN» u otro ya usado, o «➕ Nuevo cliente…». "
            "El nombre del cliente reemplaza a «EPEN» en todas las pantallas y en el Excel."
            if admin else "Elegí uno de los proyectos ya cargados.")).pack(anchor="w", padx=22)
        estado = "normal" if admin else "readonly"
        ttk.Label(self, text="Proyecto", style="Hint.TLabel").pack(anchor="w", padx=22, pady=(12, 0))
        ult = db.get_config(con, "ultimo_proyecto")
        self.v_proy = tk.StringVar(value=ult)
        proys = db.proyectos(con)
        self.cb_p = ttk.Combobox(self, textvariable=self.v_proy, values=opciones_proyectos(con, admin),
                                 state=estado, width=48)
        self.cb_p.pack(anchor="w", padx=22)
        ttk.Label(self, text="Cliente", style="Hint.TLabel").pack(anchor="w", padx=22, pady=(10, 0))
        self.v_cli = tk.StringVar(value=db.cliente_de(con, ult) if ult else "")
        self.cb_c = ttk.Combobox(self, textvariable=self.v_cli, values=opciones_clientes(con, admin), state=estado, width=48)
        self.cb_c.pack(anchor="w", padx=22)
        self.cb_p.bind("<<ComboboxSelected>>", lambda _: self._autocompletar())
        self.cb_p.bind("<FocusOut>", lambda _: self._autocompletar())
        self.cb_c.bind("<<ComboboxSelected>>", lambda _: self._cliente_elegido())
        if not admin and not v_hay(proys):
            self.lbl_err.config(text="Todavía no hay proyectos cargados. Pedile al administrador que los cree.")
        self.lbl_err.pack(anchor="w", padx=22, pady=(8, 0))
        bt = ttk.Frame(self)
        bt.pack(fill="x", padx=22, pady=10)
        ttk.Button(bt, text="Entrar", style="Accent.TButton", command=self._entrar).pack(side="right")
        ttk.Button(bt, text="Salir", command=self.destroy).pack(side="right", padx=8)
        self.bind("<Return>", lambda _: self._entrar())
        self.esperar()

    def _cliente_elegido(self):
        if self.v_cli.get() == NUEVO_CLI:
            n = pedir_nombre(self, "Nuevo cliente", "Nombre del nuevo cliente:")
            self.v_cli.set(n)

    def _autocompletar(self):
        p = self.v_proy.get().strip()
        if p == NUEVO_PROY:
            if getattr(self, "_pidiendo", False):  # al abrirse el cuadro, el combo pierde el foco y dispara esto de nuevo
                return
            self._pidiendo = True
            try:
                self.v_proy.set(pedir_nombre(self, "Nuevo proyecto", "Nombre del nuevo proyecto:"))
            finally:
                self._pidiendo = False
            return
        if p in db.proyectos(self.con):
            self.v_cli.set(db.cliente_de(self.con, p))

    def _entrar(self):
        p, c = self.v_proy.get().strip(), self.v_cli.get().strip()
        if self.rol == "admin":
            if not p or not c:
                self.lbl_err.config(text="Completá el proyecto y el cliente.")
                return
        elif p not in db.proyectos(self.con):
            self.lbl_err.config(text="Elegí un proyecto de la lista.")
            return
        self.resultado = (p, c)
        self.destroy()

    return bool(lista)

class App(tk.Tk):
    def __init__(self):
        super().__init__()
        self.title("Control de documentos")
        self.geometry("1320x830")
        self.minsize(1100, 700)
        self.configure(bg=GRIS_FONDO)
        self.con = db.conectar()
        self.sel_id = None
        self.rol, self.usuario, self.ok = "consulta", "", False
        self.v_proy, self.v_cli, self.v_lista = tk.StringVar(), tk.StringVar(), tk.StringVar()
        self._estilos()
        self.withdraw()
        if not self._ingresar():
            self.destroy()
            return
        self.ok = True
        self._armar_todo()
        self.deiconify()

    # ============================================================ ingreso y permisos
    def _admin(self):
        if self.rol != "admin":
            messagebox.showinfo("Sin permiso", "Tu usuario es de solo lectura: solo puede ver la información.")
            return False
        return True

    def _ingresar(self):
        """Primer uso (crear contraseñas) → login → elegir proyecto y cliente."""
        db.asegurar_usuarios_iniciales(self.con)
        login = LoginDialog(self, self.con)
        if login.rol is None:
            return False
        self.rol, self.usuario = login.rol, login.usuario
        ctx = ContextoDialog(self, self.con, self.rol)
        if not ctx.resultado:
            return False
        proy, cli = ctx.resultado
        if self.rol == "admin":
            db.asegurar_proyecto(self.con, proy, cli)
            db.set_cliente_proyecto(self.con, proy, cli)
        db.set_config(self.con, "ultimo_proyecto", proy)
        self.v_proy.set(proy)
        return True

    def cerrar_sesion(self):
        self.withdraw()
        if not self._ingresar():
            self.destroy()
            return
        self._armar_todo()
        self.deiconify()

    def usuarios(self):
        if self._admin():
            UsuariosDialog(self, self.con)

    # ============================================================ armado de pantallas
    def _armar_todo(self, tab=None):
        """(Re)construye cabecera y pestañas. Se usa al ingresar y cada vez que cambia el cliente,
        porque el nombre del cliente reemplaza a «EPEN» en todos los textos."""
        global NOMBRE_CLI
        for w in self.winfo_children():
            w.destroy()
        proy = self.v_proy.get().strip()
        NOMBRE_CLI = db.cliente_de(self.con, proy) or "EPEN"
        self.v_cli.set(db.cliente_de(self.con, proy))
        self._proy_actual = proy
        self._armar_cabecera()
        self.nb = ttk.Notebook(self)
        self.nb.pack(fill="both", expand=True, padx=10, pady=(0, 10))
        self.tab_carga = ttk.Frame(self.nb)
        self.tab_cons = ttk.Frame(self.nb)
        self.tab_mod = ttk.Frame(self.nb)
        if self.rol == "admin":
            self.nb.add(self.tab_carga, text="  1 · Cargar documentos  ")
        if self.rol != "gerencia":
            self.nb.add(self.tab_cons, text="  2 · Consultar documento  ")
        if self.rol == "admin":
            self.nb.add(self.tab_mod, text="  3 · Modificar / Actualizar  ")
        self.tab_dash = ttk.Frame(self.nb)
        self.nb.add(self.tab_dash, text="  4 · Dashboard  ")
        self.tab_kpi = ttk.Frame(self.nb)
        self.nb.add(self.tab_kpi, text="  5 · KPI  ")
        self._armar_carga()
        self._armar_consulta()
        self._armar_modificar()
        self._armar_dashboard()
        self._armar_kpi()
        self.nb.bind("<<NotebookTabChanged>>", self._cambio_tab)
        self.cambiar_proyecto(inicial=True)
        if tab is None:
            tab = ("dash" if self.rol == "gerencia"
                   else "cons" if db.buscar(self.con, "", self.v_proy.get().strip()) else "carga")
        frame = {"carga": self.tab_carga, "cons": self.tab_cons, "mod": self.tab_mod, "dash": self.tab_dash, "kpi": self.tab_kpi}[tab]
        self.nb.select(frame if str(frame) in self.nb.tabs()
                       else self.tab_dash if self.rol == "gerencia" else self.tab_cons)

    def _reconstruir(self):
        sel = self.nb.select()
        tab = ("mod" if sel == str(self.tab_mod) else "carga" if sel == str(self.tab_carga)
               else "dash" if sel == str(self.tab_dash) else "kpi" if sel == str(self.tab_kpi) else "cons")
        self.after_idle(lambda: self._armar_todo(tab))

    def _leyenda(self, parent, filtrable=False):
        """Explica los colores de las filas: salen del estado cargado (el más avanzado entre cliente y {cli})."""
        fr = ttk.Frame(parent)
        fr.pack(anchor="w", padx=4, pady=(0, 2))
        ttk.Label(fr, text="Colores:", style="Hint.TLabel").pack(side="left", padx=(0, 4))
        if filtrable:
            self._etiquetas_filtro = {}
        for clase, txt in (("ok", "Aprobado / APC"), ("obs", "Aprobado con observaciones"), ("mal", "DPC / rechazado"),
                           ("rev", "En revisión"), ("nobs", "No se observa"), ("visto", "Visto"),
                           ("vacio", "Sin estado")):
            lb = tk.Label(fr, text=f" {txt} ", bg=COLORES[clase][0] or "white", fg="#333", font=(FUENTE, 8),
                          relief="solid", bd=1)
            lb.pack(side="left", padx=2)
            if filtrable:  # clic: muestra solo los documentos en ese estado; otro clic lo saca
                lb.config(cursor="hand2")
                lb.bind("<Button-1>", lambda _e, c=clase: self._filtrar_estado(c))

        if filtrable:
            ttk.Label(fr, text="  (clic en un color para filtrar)", style="Hint.TLabel").pack(side="left")

        self.filtro_estado = None if getattr(self, "filtro_estado", None) == clase else clase
        for c, lb in self._etiquetas_filtro.items():
            activo = c == self.filtro_estado
            lb.config(font=(FUENTE, 8, "bold" if activo else "normal"), relief="sunken" if activo else "solid",
                      bd=3 if activo else 1)
        self._llenar_consulta()

    def n_visto(self):

            return int(self.v_lista.get().split("N°")[1].split()[0])
        except (IndexError, ValueError):
            return self.n_actual()

    def es_actual(self):
        return self.n_visto() == self.n_actual()

    def cambiar_proyecto(self, inicial=False):
        """Cambia de proyecto (el administrador puede escribir uno nuevo; se crea al apretar Enter)."""
        proy = self.v_proy.get().strip()
        if proy == NUEVO_PROY:
            proy = pedir_nombre(self, "Nuevo proyecto", "Nombre del nuevo proyecto:")
            self.v_proy.set(proy or self._proy_actual)
            proy = self.v_proy.get().strip()
            if not proy or proy == self._proy_actual:
                return
        todos = db.proyectos(self.con)
        if not inicial:
            if not proy:
                self.v_proy.set(self._proy_actual)
                return
            if proy not in todos:
                if self.rol != "admin":
                    self.v_proy.set(self._proy_actual)
                    return
                db.asegurar_proyecto(self.con, proy, self.v_cli.get().strip())
                todos = db.proyectos(self.con)
            if proy != self._proy_actual:
                db.set_config(self.con, "ultimo_proyecto", proy)
                if (db.cliente_de(self.con, proy) or "EPEN") != NOMBRE_CLI or self.v_cli.get() != db.cliente_de(self.con, proy):
                    self._reconstruir()  # otro cliente: se renombran los textos
                    return
        self._proy_actual = proy
        adm = self.rol == "admin"
        if adm:
            self.after(150, self._sincronizar_excel)
        self.cb_proy["values"] = opciones_proyectos(self.con, adm)
        self.cb_cli["values"] = opciones_clientes(self.con, adm)
        self.v_cli.set(db.cliente_de(self.con, proy))
        self._cargar_listas()
        self.refrescar()

    def cambiar_cliente(self):
        """Cambia el cliente del proyecto: el nuevo nombre reemplaza a «EPEN» en todos lados."""
        proy, cli = self.v_proy.get().strip(), self.v_cli.get().strip()
        actual = db.cliente_de(self.con, proy)
        if cli == NUEVO_CLI:
            cli = pedir_nombre(self, "Nuevo cliente", "Nombre del nuevo cliente:")
            self.v_cli.set(cli or actual)
        if self.rol != "admin" or not proy or cli == actual:
            self.v_cli.set(actual)
            return
        if not cli:
            self.v_cli.set(actual)
            return
        if actual and not messagebox.askyesno(
                "Cambiar cliente", f"Vas a cambiar el cliente del proyecto «{proy}» de «{actual}» a «{cli}».\n"
                                   f"El nombre «{cli}» va a reemplazar a «{actual}» en todas las pantallas y "
                                   "en el Excel que se genere.\n\n¿Continuar?"):
            self.v_cli.set(actual)
            return
        db.set_cliente_proyecto(self.con, proy, cli)
        self._reconstruir()

    def _cargar_listas(self):
        n = self.n_actual()
        self.cb_lista["values"] = [f"N°{n} (actual)"] + [f"N°{i}" for i in range(n - 1, 0, -1)]
        self.v_lista.set(f"N°{n} (actual)")
        self.btn_emitir.config(text=f"Nueva revisión N°{n + 1} (se entregaron archivos)")

    def _sincronizar_excel(self, manual=False):
        """Compara con el Excel vinculado: si se borraron hojas Rev (N) del final, la app vuelve a esa revisión."""
        if self.rol != "admin":
            return False
        proy = self.v_proy.get().strip()
        ruta = db.excel_de(self.con, proy)
        if not proy or not ruta or not os.path.exists(ruta):
            if manual:
                messagebox.showinfo("Sincronizar", "No hay un Excel vinculado (o no lo encuentro).")
            return False
        hojas = db.hojas_excel(ruta)
        if hojas is None:
            if manual:
                messagebox.showwarning("Sincronizar", "No pude leer el Excel vinculado.")
            return False
        actual = self.n_actual()
        if not hojas or hojas[-1] >= actual:
            if manual:
                messagebox.showinfo("Sincronizar", "La app y el Excel están al día: coinciden las revisiones.")
            return False
        ultima = hojas[-1]
        if not messagebox.askyesno(
                "El Excel cambió",
                f"La app está en la revisión N°{actual}, pero la última hoja del Excel vinculado es Rev ({ultima}).\n\n"
                f"¿Volver a la revisión N°{ultima}? Se descartan las revisiones posteriores de la app (los documentos "
                "conservan su estado actual) y antes se hace una copia de seguridad de la base."):
            return False
        copia = db.retroceder_a_excel(self.con, proy, ultima)
        self._cargar_listas()
        self.refrescar()
        messagebox.showinfo("Sincronizado", f"Ahora trabajás sobre la revisión N°{ultima}.\n\nCopia de seguridad:\n{copia}")
        return True

    def guardar_en_excel(self):
        """Pasa el estado actual de los documentos a la última revisión del Excel vinculado (modifica la hoja
        Rev (N) en el lugar, sin agregar hojas) y la deja registrada en el historial. Devuelve un texto de estado."""
        proy = self.v_proy.get().strip()
        n = self.n_actual()
        db.guardar_lista(self.con, proy, n)
        ruta = db.excel_de(self.con, proy)
        if not ruta or not os.path.exists(ruta):
            return ""
        primera = ruta not in getattr(self, "_respaldados", set())  # una sola copia de respaldo por sesión
        try:
            db.actualizar_hoja_excel(self.con, ruta, proy, n, respaldar=primera)
        except PermissionError:
            messagebox.showwarning("Excel abierto", "Guardé los cambios en la app, pero no pude modificar el Excel porque "
                                                    "está abierto.\nCerralo y volvé a guardar: se actualiza todo de una vez.")
            return "⚠ guardado en la app; Excel sin actualizar"
        except Exception as e:  # noqa: BLE001
            messagebox.showwarning("Excel", f"Guardé los cambios en la app, pero no pude actualizar el Excel:\n{e}")
            return "⚠ guardado en la app; Excel sin actualizar"
        self._respaldados = getattr(self, "_respaldados", set()) | {ruta}
        return f"· Excel Rev ({n}) actualizado"

    def emitir(self):
        """Abre una revisión nueva (cuando se entregan archivos): agrega la hoja Rev (N+1) al Excel vinculado."""
        if not self._admin():
            return
        self._sincronizar_excel()
        proy = self.v_proy.get().strip()
        n = self.n_actual()
        if not db.buscar(self.con, "", proy):
            messagebox.showinfo("Nueva revisión", "Todavía no hay documentos en este proyecto.")
            return
        ruta = db.excel_de(self.con, proy)
        if ruta and not os.path.exists(ruta):
            messagebox.showwarning("Excel no encontrado", f"No encuentro el Excel vinculado:\n{ruta}\n"
                                                          "Después te pido que elijas otro.")
            ruta = ""
        msg = (f"Se cierra la Revisión N°{n} y se abre la N°{n + 1}, que arranca con el estado actual de los documentos.\n"
               "Usá esto solo cuando se entregan archivos nuevos: mientras tanto, los cambios se guardan en la "
               f"revisión N°{n}.\n\n")
        if ruta:
            msg += (f"Se agrega la hoja «Rev ({n + 1})» al Excel de control:\n{ruta}\n"
                    "(hago una copia de respaldo antes; el Excel tiene que estar cerrado)\n\n")
        if not messagebox.askyesno("Nueva revisión", msg + "¿Continuar?"):
            return
        if ruta:
            try:
                db.actualizar_hoja_excel(self.con, ruta, proy, n, respaldar=False)  # deja Rev (N) al día
                respaldo = db.agregar_hoja_excel(self.con, ruta, proy, n + 1)
            except PermissionError:
                messagebox.showerror("No se pudo modificar el Excel",
                                     "No pude escribir en el Excel: probablemente está abierto.\n"
                                     "Cerralo y volvé a intentar. La revisión NO se creó.")
                return
            except Exception as e:  # noqa: BLE001
                messagebox.showerror("No se pudo modificar el Excel", f"{e}\n\nLa revisión NO se creó.")
                return
            aviso = f"\n\nSe agregó la hoja «Rev ({n + 1})» al Excel.\nCopia de respaldo:\n{respaldo}"
        else:
            aviso = ""
        db.emitir_lista(self.con, proy)
        self._cargar_listas()
        self.refrescar()
        messagebox.showinfo("Nueva revisión", f"Ahora trabajás sobre la Revisión N°{n + 1}.{aviso}")

    def _contexto(self):
        proy, cli = self.v_proy.get().strip(), self.v_cli.get().strip()
        if not proy:
            messagebox.showwarning("Falta el proyecto", "Escribí o elegí el Proyecto en la parte de arriba.")
            return None
        return proy, cli

    def _post_carga(self, proy):
        self.cambiar_proyecto()
        self.v_proy.set(proy)

    def agregar_uno(self):
        if not self._admin():
            return
        ctx = self._contexto()
        if not ctx:
            return
        d = {k: v.get().strip() for k, v in self.nv.items()}
        if not (d["titulo"] or d["cod_epen"] or d["cod_cliente"]):
            messagebox.showwarning("Faltan datos", "Cargá al menos el título o un código.")
            return
        d.update(proyecto=ctx[0], cliente=ctx[1], entregado=0, nuestro=int(self.v_nuestro.get()))
        doc_id, creado = db.agregar(self.con, d)
        if not creado:
            messagebox.showinfo("Ya existe", "Ese documento ya está cargado en este proyecto.")
            return
        for v in self.nv.values():
            v.set("")
        self._post_carga(ctx[0])
        if self.v_nuestro.get():
            self.seleccionar(doc_id)

    def agregar_masa(self):
        if not self._admin():
            return
        ctx = self._contexto()
        if not ctx:
            return
        nuevos = repes = 0
        for linea in self.txt_masa.get("1.0", "end").splitlines():
            if not linea.strip():
                continue
            p = [x.strip() for x in linea.split("\t")]
            cc, ce, t = (p + ["", "", ""])[:3] if len(p) >= 3 else (
                ("", p[0], p[1]) if len(p) == 2 else ("", "", p[0]))
            _, creado = db.agregar(self.con, dict(proyecto=ctx[0], cliente=ctx[1], cod_cliente=cc,
                                                  cod_epen=ce, titulo=t, entregado=0,
                                                  nuestro=int(self.v_nuestro.get())))
            nuevos += creado
            repes += not creado
        self.txt_masa.delete("1.0", "end")
        self._post_carga(ctx[0])
        messagebox.showinfo("Carga terminada", f"Documentos nuevos: {nuevos}\nYa existentes (omitidos): {repes}")
        if nuevos and self.v_nuestro.get():
            self.nb.select(self.tab_cons)

    @staticmethod
    def _f(fecha):
        return fecha.strftime("%d/%m/%Y") if fecha else "—"

    def _texto_kpi(self, d):
        """Una línea con lo que miden los KPI para este documento: plazo de entrega y ciclos de corrección."""
        ev = kpi.evaluar(d)
        partes = [f"📅 Fecha límite: {self._f(ev['plan'])}", f"Entrega real: {self._f(ev['real'])}"]
        if ev["desvio"] is not None:
            partes[-1] += (f" ({abs(ev['desvio'])} días de atraso)" if ev["desvio"] > 0
                           else f" ({abs(ev['desvio'])} días antes)" if ev["desvio"] < 0 else " (el día límite)")
        partes.append(f"Plazo: {ev['plazo']}")
        if d["transmittal"]:
            partes.append(f"Transmittal: {d['transmittal']}")
        txt = "   ·   ".join(partes)
        for c in db.ciclos_doc(self.con, d["id"]):
            dev, rei = db.parse_fecha(c["fecha_devolucion"]), db.parse_fecha(c["fecha_reenvio"])
            if dev and rei:
                detalle = f"devuelta {self._f(dev)}, reenviada {self._f(rei)} ({(rei - dev).days} días)"
            elif dev:
                detalle = f"devuelta {self._f(dev)}, pendiente de reenvío"
            else:
                detalle = f"reenviada {self._f(rei)}" if rei else "sin fechas cargadas"
            txt += f"\n🔁 Corrección Rev {c['rev'] or '—'}: {detalle}"
        return txt

    @staticmethod
    def _torta(cv, titulo, partes):
        cv.delete("all")
        w, h = max(cv.winfo_width(), 300), max(cv.winfo_height(), 260)
        cv.create_text(w / 2, 24, text=titulo, font=(FUENTE, 12, "bold"), fill=AZUL_OSC)
        total = sum(n for _, n, _ in partes)
        if total == 0:
            cv.create_text(w / 2, h / 2, text="Sin datos", fill=GRIS_TXT, font=(FUENTE, 11))
            return
        d = max(min(h - 80, w * 0.46), 110)
        x0, y0 = 20, 52
        caja = (x0, y0, x0 + d, y0 + d)
        cx, cy, ang = x0 + d / 2, y0 + d / 2, 90.0
        for nombre, n, color in partes:
            if n == 0:
                continue
            ext = 360 * n / total
            if ext >= 359.9:
                cv.create_oval(caja, fill=color, outline="white", width=2)
            else:
                cv.create_arc(caja, start=ang, extent=-ext, fill=color, outline="white", width=2)
            if ext >= 20:
                mid = math.radians(ang - ext / 2)
                cv.create_text(cx + d * 0.32 * math.cos(mid), cy - d * 0.32 * math.sin(mid),
                               text=f"{round(100 * n / total)}%", font=(FUENTE, 11, "bold"),
                               fill="#1F2937" if color == "#B8C0CC" else "white")
            ang -= ext
        lx, ly = x0 + d + 22, y0 + 4
        for nombre, n, color in partes:
            cv.create_rectangle(lx, ly, lx + 14, ly + 14, fill=color, outline="")
            cv.create_text(lx + 22, ly + 7, anchor="w", font=(FUENTE, 10),
                           text=f"{nombre}: {n} ({round(100 * n / total)}%)")
            ly += 26

    def _llenar_lista_mod(self):
        if not hasattr(self, "tree_m"):
            return
        self.tree_m.delete(*self.tree_m.get_children())
        for d in db.buscar(self.con, self.v_buscar_m.get(), self.v_proy.get().strip()):
            self.tree_m.insert("", "end", iid=str(d["id"]), tags=(clase_doc(d),),
                               values=(d["cod_epen"] or d["cod_cliente"], d["titulo"]))
        if self.sel_id and self.tree_m.exists(str(self.sel_id)):
            self.tree_m.selection_set(str(self.sel_id))

    def _sel_modificar(self, _=None):
        s = self.tree_m.selection()
        if s and int(s[0]) != self.sel_id:
            self.sel_id = int(s[0])
        if s:
            self._cargar_form()

    def _cambio_tab(self, _=None):
        if self.nb.select() == str(self.tab_mod):
            self._llenar_lista_mod()
            if self.sel_id and self.tree_m.exists(str(self.sel_id)):
                self.tree_m.selection_set(str(self.sel_id))
                self.tree_m.see(str(self.sel_id))
                self._cargar_form()
        elif self.nb.select() == str(self.tab_cons):
            self._llenar_consulta()
        elif self.nb.select() == str(self.tab_dash):
            self._llenar_dashboard()
        elif self.nb.select() == str(self.tab_kpi):
            self._llenar_kpi()

    def _cargar_form(self):
        d = db.get_doc(self.con, self.sel_id) if self.sel_id else None
        if not d:
            return
        n = self.n_actual()
        self.lbl_doc_m.config(text=f"{d['cod_epen'] or d['cod_cliente']}  —  {d['titulo']}"
                                   f"      (Revisión N°{n} de la lista)")
        for k in self.rv:
            self.rv[k].set(d[k])
        self.v_entregado.set(bool(d["entregado"]))
        for k, ki in (("estado_cliente", "inf_cliente"), ("estado_epen", "inf_epen")):
            base, inf = db.partir_estado(d[k])
            self.ev[k].set(base)
            self.ev[ki].set(inf)
        self.txt_com.delete("1.0", "end")
        self.txt_com.insert("1.0", d["comentarios"])
        self.fv["fecha_limite_go"].set(d["fecha_limite_go"])
        self.fv["fecha_real"].set(d["fecha_real"])
        self.fv["transmittal"].set(d["transmittal"])
        revs = [c["rev"] for c in db.ciclos_doc(self.con, d["id"])]
        if "dpc" in db._norm(d["estado_epen"]) and d["rev_epen"] and d["rev_epen"] not in revs:
            revs.append(d["rev_epen"])  # devuelto ahora con DPC: todavía no tiene fechas cargadas
        self.cb_ciclo["values"] = revs
        self.fv["ciclo_rev"].set(d["rev_epen"] if d["rev_epen"] in revs else (revs[-1] if revs else ""))
        self._cargar_ciclo()
        self.lbl_ok.config(text="")

    def _cargar_ciclo(self):
        c = db.ciclo(self.con, self.sel_id, self.fv["ciclo_rev"].get()) if self.sel_id else None
        self.fv["fecha_devolucion"].set(c["fecha_devolucion"] if c else "")
        self.fv["fecha_reenvio"].set(c["fecha_reenvio"] if c else "")

    def _al_entregar(self):
        """Al marcar «entregado» sin fecha real, propone la de hoy (se puede cambiar)."""
        if self.v_entregado.get() and not self.fv["fecha_real"].get().strip():
            self.fv["fecha_real"].set(datetime.now().strftime("%d/%m/%Y"))

    def _estado_ref(self, ref):
        cli = db.juntar_estado(self.ev["estado_cliente"].get(), self.ev["inf_cliente"].get())
        epen = db.juntar_estado(self.ev["estado_epen"].get(), self.ev["inf_epen"].get())
        return {"cliente": cli or epen, "epen": epen or cli, "ambos": epen or cli}[ref]

    def subir_rev(self, campo, ref):
        if self.sel_id is None:
            return
        actual = self.rv[campo].get().strip()
        if actual and not db.rev_valida(actual):
            messagebox.showwarning("Revisión", "La revisión debe ser una letra (A, B…) o un número (0, 1…).")
            return
        self.rv[campo].set(db.siguiente_rev(actual, self._estado_ref(ref)))

    def guardar(self):
        if not self._admin():
            return
        if self.sel_id is None:
            messagebox.showinfo("Guardar", "Elegí primero un documento de la lista.")
            return
        datos = {k: v.get().strip().upper() for k, v in self.rv.items()}
        if not all(db.rev_valida(v) for v in datos.values()):
            messagebox.showwarning("Revisión", "Las revisiones deben ser una letra (A, B…) o un número (0, 1…).")
            return
        datos["estado_cliente"] = db.juntar_estado(self.ev["estado_cliente"].get(), self.ev["inf_cliente"].get())
        datos["estado_epen"] = db.juntar_estado(self.ev["estado_epen"].get(), self.ev["inf_epen"].get())
        datos["entregado"] = int(self.v_entregado.get())
        datos["comentarios"] = self.txt_com.get("1.0", "end").strip()
        try:
            datos["fecha_limite_go"] = db.norm_fecha(self.fv["fecha_limite_go"].get())
            datos["fecha_real"] = db.norm_fecha(self.fv["fecha_real"].get())
            dev, rei = db.norm_fecha(self.fv["fecha_devolucion"].get()), db.norm_fecha(self.fv["fecha_reenvio"].get())
        except ValueError as e:
            messagebox.showwarning("Fecha", str(e))
            return
        datos["transmittal"] = self.fv["transmittal"].get().strip()
        rev_c = self.fv["ciclo_rev"].get().strip().upper()
        if (dev or rei) and not rev_c:
            messagebox.showwarning("Corrección", "Elegí la revisión devuelta para guardar las fechas de la corrección.")
            return
        db.actualizar(self.con, self.sel_id, datos)
        if rev_c:
            db.guardar_ciclo(self.con, self.sel_id, rev_c, dev, rei)
        self.cb_cli["values"] = db.clientes(self.con)
        self.refrescar()
        self._cargar_form()
        extra = self.guardar_en_excel()
        self.lbl_ok.config(text=f"✔ Guardado {datetime.now():%H:%M} {extra}")

    def eliminar(self):
        if not self._admin():
            return
        if self.sel_id is None:
            return
        if messagebox.askyesno("Eliminar", "¿Eliminar este documento y todo su historial?\nNo se puede deshacer."):
            db.eliminar(self.con, self.sel_id)
            self.sel_id = None
            self.lbl_doc_m.config(text="(elegí un documento de la lista)")
            self.refrescar()
            self._limpiar_ficha()
